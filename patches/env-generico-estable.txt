DM_PICTURE_FIT=fit

DM_TEX_LAZY=1

# CAMBIO 2 — FPS_HOLD=off (hle_nvn.c fstep_tick): un frame tardío se muestra tarde

# TEX_RELEASE=1 primero (seguro); 2 solo si no hay texturas negras (ver doc genérico).
DM_TEX_RELEASE=1
